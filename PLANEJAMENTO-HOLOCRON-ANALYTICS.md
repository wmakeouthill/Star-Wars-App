### Implementação Técnica do Quiz

```python
# domain/entities/quiz.py
from dataclasses import dataclass, field
from typing import List, Optional
from enum import Enum
from datetime import datetime

class QuizDifficulty(str, Enum):
    YOUNGLING = "youngling"
    PADAWAN = "padawan"
    KNIGHT = "knight"
    MASTER = "master"
    SITH_LORD = "sith_lord"

@dataclass
class QuizQuestion:
    id: str
    text: str
    options: List[str]
    correct_answer: int
    difficulty: QuizDifficulty
    category: str  # characters, planets, starships, films
    explanation: str
    related_data: dict  # Dados da SWAPI usados
    time_limit: int  # segundos

@dataclass
class QuizSession:
    id: str
    user_id: str
    difficulty: QuizDifficulty
    questions: List[QuizQuestion]
    current_index: int = 0
    score: int = 0
    streak: int = 0
    max_streak: int = 0
    answers: List[dict] = field(default_factory=list)
    started_at: datetime = field(default_factory=datetime.utcnow)
    completed_at: Optional[datetime] = None

# application/services/quiz_service.py
class QuizService:
    """Serviço de Quiz com geração dinâmica via IA."""
    
    def __init__(
        self, 
        swapi_client: SWAPIClient,
        ai_service: YodaAIService,
        user_repository: IUserRepository
    ):
        self._swapi = swapi_client
        self._ai = ai_service
        self._users = user_repository
    
    async def start_quiz(
        self, 
        user_id: str, 
        difficulty: QuizDifficulty,
        category: Optional[str] = None,
        num_questions: int = 10
    ) -> QuizSession:
        """Inicia uma nova sessão de quiz."""
        
        # Busca dados relevantes da SWAPI
        data_pool = await self._fetch_quiz_data(category)
        
        # Gera perguntas dinamicamente com IA
        questions = await self._ai.generate_quiz_questions(
            data=data_pool,
            difficulty=difficulty,
            count=num_questions
        )
        
        session = QuizSession(
            id=generate_uuid(),
            user_id=user_id,
            difficulty=difficulty,
            questions=questions
        )
        
        return session
    
    async def answer_question(
        self,
        session: QuizSession,
        answer_index: int,
        time_taken: float
    ) -> QuizAnswerResult:
        """Processa resposta e retorna feedback."""
        
        question = session.questions[session.current_index]
        is_correct = answer_index == question.correct_answer
        
        # Calcula XP com multiplicadores
        base_xp = self._get_base_xp(session.difficulty)
        time_bonus = self._calculate_time_bonus(time_taken, question.time_limit)
        streak_multiplier = 1 + (session.streak * 0.1) if is_correct else 1
        
        xp_earned = int(base_xp * streak_multiplier + time_bonus) if is_correct else 0
        
        # Atualiza streak
        if is_correct:
            session.streak += 1
            session.max_streak = max(session.max_streak, session.streak)
        else:
            session.streak = 0
        
        # Gera feedback do Yoda
        feedback = await self._ai.generate_quiz_feedback(
            question=question,
            is_correct=is_correct,
            user_answer=answer_index
        )
        
        return QuizAnswerResult(
            is_correct=is_correct,
            correct_answer=question.correct_answer,
            explanation=question.explanation,
            yoda_feedback=feedback,
            xp_earned=xp_earned,
            streak=session.streak,
            time_bonus=time_bonus
        )
```

```python
from fastapi import APIRouter, Depends, WebSocket

router = APIRouter(prefix="/quiz", tags=["Jedi Trials - Quiz"])

@router.post("/start", response_model=QuizSessionResponse)
async def start_quiz(
    difficulty: QuizDifficulty,
    category: Optional[str] = None,
    num_questions: int = Query(10, ge=5, le=50),
    service: QuizService = Depends(get_quiz_service),
    current_user: User = Depends(get_current_user)
):
    """
    Inicia uma nova sessão de quiz.
    
    🎮 Modos: youngling, padawan, knight, master, sith_lord
    """
    return await service.start_quiz(
        user_id=current_user.id,
        difficulty=difficulty,
        category=category,
        num_questions=num_questions
    )

@router.post("/{session_id}/answer", response_model=QuizAnswerResult)
async def answer_question(
    session_id: str,
    answer: QuizAnswerRequest,
    service: QuizService = Depends(get_quiz_service)
):
    """Envia resposta para a pergunta atual."""
    return await service.answer_question(
        session_id=session_id,
        answer_index=answer.selected_option,
        time_taken=answer.time_taken
    )
```

### ⚠️ Problema: A SWAPI Não Retorna Atributos de RPG

A API do Star Wars (SWAPI) retorna apenas dados básicos:

```json
{
  "name": "Luke Skywalker",
  "height": "172",
  "mass": "77",
  "hair_color": "blond",
  "skin_color": "fair",
  "eye_color": "blue",
  "birth_year": "19BBY",
  "gender": "male",
  "homeworld": "https://swapi.dev/api/planets/1/",
  "films": ["https://swapi.dev/api/films/1/", ...],
  "species": [...],
  "vehicles": [...],
  "starships": [...]
}
```

**Não há:** força, agilidade, poder da Força, facção, raridade, etc.

### 💡 Solução: Sistema Híbrido de Enriquecimento de Dados

Vamos usar **3 estratégias combinadas** para criar os atributos:

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                    PIPELINE DE ENRIQUECIMENTO DE DADOS                      │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│   SWAPI (Dados Brutos)          NOSSA BASE              VERTEX AI           │
│   ┌─────────────────┐     ┌─────────────────┐     ┌─────────────────┐      │
│   │ • name          │     │ • stats (RPG)   │     │ • Gera stats    │      │
```

```python
# infrastructure/external/vertex_ai/stats_generator.py
class CharacterStatsGenerator:
    """Gera stats de RPG usando IA baseado no lore de Star Wars."""
    
    PROMPT_TEMPLATE = """
    Você é um especialista em Star Wars. Baseado no seu conhecimento do lore,
    gere atributos de RPG (0-100) para o personagem abaixo.
    
    PERSONAGEM: {name}
    DADOS CONHECIDOS:
    - Altura: {height} cm
    - Peso: {mass} kg
    - Planeta natal: {homeworld}
    - Aparece nos filmes: {films}
    - Espécie: {species}
    
    ATRIBUTOS A GERAR:
    1. strength (força física): 0-100
    2. agility (agilidade/velocidade): 0-100
    3. intelligence (inteligência/sabedoria): 0-100
    4. resistance (resistência/vida): 0-100
    5. force_power (poder da Força): 0-100 (0 se não for sensitivo à Força)
    
    TAMBÉM GERE:
    - rarity: common/uncommon/rare/epic/legendary (baseado na importância na saga)
    - faction: lista de facções (Jedi, Sith, Rebel, Empire, etc.)
    - title: título do personagem (ex: "Cavaleiro Jedi", "Caçador de Recompensas")
    - quote: frase icônica do personagem (se existir)
    - special_ability: nome e descrição de uma habilidade especial temática
    
    RESPONDA APENAS EM JSON VÁLIDO.
    """
    
    async def generate_stats(self, character_data: dict) -> CharacterStats:
        """Gera stats para um personagem usando IA."""
        
        # Primeiro, verifica se já temos na base
        if character_data["name"] in CHARACTER_STATS_DATABASE:
            return CHARACTER_STATS_DATABASE[character_data["name"]]
        
        # Se não, gera com IA
        prompt = self.PROMPT_TEMPLATE.format(
            name=character_data["name"],
            height=character_data.get("height", "desconhecido"),
            mass=character_data.get("mass", "desconhecido"),
            homeworld=character_data.get("homeworld_name", "desconhecido"),
            films=", ".join(character_data.get("film_names", [])),
            species=character_data.get("species_name", "Humano")
        )
        
        response = await self._model.generate_content_async(prompt)
        stats_json = self._parse_json_response(response.text)
        
        # Cache o resultado para não gerar novamente
        await self._cache.set(f"stats:{character_data['name']}", stats_json)
        
        return self._convert_to_stats(stats_json)
```

### Estratégia 3: Cálculos Derivados dos Dados da SWAPI

Alguns atributos podem ser estimados a partir dos dados disponíveis:

```python
# application/services/stats_calculator.py
class StatsCalculator:
    """Calcula estimativas de stats baseado nos dados da SWAPI."""
    
    def estimate_from_swapi_data(self, character: dict) -> dict:
        """Estima alguns atributos baseado nos dados disponíveis."""
        
        estimates = {}
        
        # Força física baseada em altura e massa
        height = self._parse_number(character.get("height"))
        mass = self._parse_number(character.get("mass"))
        
        if height and mass:
            # Personagens maiores e mais pesados tendem a ser mais fortes
            bmi = mass / ((height / 100) ** 2)
            if bmi > 30 and height > 200:  # Grande e pesado (Chewbacca)
                estimates["strength_hint"] = "high"
            elif height < 100:  # Pequeno (Yoda)
                estimates["strength_hint"] = "low"
        
        # Número de filmes indica importância (raridade)
        num_films = len(character.get("films", []))
        if num_films >= 5:
            estimates["rarity_hint"] = "legendary"
        elif num_films >= 3:
            estimates["rarity_hint"] = "epic"
        elif num_films >= 2:
            estimates["rarity_hint"] = "rare"
        else:
            estimates["rarity_hint"] = "common"
        
        # Número de naves/veículos pode indicar agilidade/pilotagem
        num_vehicles = len(character.get("vehicles", [])) + len(character.get("starships", []))
        if num_vehicles >= 3:
            estimates["agility_hint"] = "high"
        
        return estimates
```

```python
class CharacterEnrichmentService:
    """Serviço que enriquece dados da SWAPI com stats de RPG."""
    
    def __init__(
        self,
        swapi_client: SWAPIClient,
        stats_db: CharacterStatsDatabase,
        ai_generator: CharacterStatsGenerator,
        calculator: StatsCalculator
    ):
        self._swapi = swapi_client
        self._stats_db = stats_db
        self._ai = ai_generator
        self._calc = calculator
    
    async def get_enriched_character(self, character_id: str) -> CharacterCard:
        """Busca personagem e enriquece com todos os dados."""
        
        # 1. Busca dados brutos da SWAPI
        swapi_data = await self._swapi.get_character(character_id)
        
        # 2. Resolve referências (homeworld, films, species)
        swapi_data = await self._resolve_references(swapi_data)
        
        # 3. Tenta buscar stats da nossa base de dados
        stats = self._stats_db.get(swapi_data["name"])
        
        if not stats:
            # 4. Se não encontrou, calcula hints dos dados SWAPI
            hints = self._calc.estimate_from_swapi_data(swapi_data)
            
            # 5. Gera stats completos com IA (usando hints como contexto)
            stats = await self._ai.generate_stats(swapi_data, hints)
        
        # 6. Monta o CharacterCard completo
        return CharacterCard(
            # Dados da SWAPI
            id=character_id,
            name=swapi_data["name"],
            height=swapi_data.get("height"),
            mass=swapi_data.get("mass"),
            birth_year=swapi_data.get("birth_year"),
            homeworld=swapi_data.get("homeworld_name"),
            films=swapi_data.get("film_names", []),
            
            # Dados enriquecidos
            stats=stats["stats"],
            rarity=stats["rarity"],
            faction=stats["faction"],
            title=stats["title"],
            quote=stats["quote"],
            special_ability=stats["special_ability"]
        )
```

```python
    @property
    def overall_rating(self) -> int:
        """Rating geral do personagem."""
        return (
            self.strength + 
            self.agility + 
            self.intelligence + 
            self.resistance +

    quote: str  # Frase icônica do personagem
    lore_snippet: str  # Breve história

# application/services/character_card_service.py
class CharacterCardService:
    """Serviço para gerar cards de personagens."""
    
    STATS_MAPPING = {
        # Personagens com stats conhecidos baseados no lore
        "Luke Skywalker": CharacterStats(75, 80, 85, 70, 95),
        "Darth Vader": CharacterStats(85, 62, 75, 92, 98),
        "Yoda": CharacterStats(45, 70, 99, 60, 100),
        "Han Solo": CharacterStats(70, 85, 75, 68, 0),
        "Chewbacca": CharacterStats(95, 60, 65, 90, 0),
        "Leia Organa": CharacterStats(55, 70, 90, 65, 75),
        "Obi-Wan Kenobi": CharacterStats(78, 82, 90, 74, 85),
        "Palpatine": CharacterStats(40, 45, 95, 55, 99),
        "Boba Fett": CharacterStats(80, 88, 78, 75, 0),
        "R2-D2": CharacterStats(20, 50, 85, 95, 0),
    }
    
    async def generate_card(self, character_id: str) -> CharacterCard:
        """Gera card completo de um personagem."""
        
        # Busca dados da SWAPI
        character = await self._swapi.get_character(character_id)
        
        # Calcula ou busca stats
        if character.name in self.STATS_MAPPING:
            stats = self.STATS_MAPPING[character.name]
        else:
            stats = await self._ai.calculate_character_stats(character)
        
        # Determina raridade
        rarity = self._calculate_rarity(character, stats)
        
        # Gera habilidade especial e quote com IA
        special = await self._ai.generate_special_ability(character)
        quote = await self._ai.get_iconic_quote(character)
        
        return CharacterCard(
            id=character_id,
            name=character.name,
            stats=stats,
            rarity=rarity,
            special_ability=special,
            quote=quote,
            # ... outros campos
        )
```

```typescript
    <S.Container>
      <S.Header>
        <RankBadge rank={profile.jediRank} />

        </S.UserInfo>
      </S.Header>

        current={profile.totalXp} 
        toNextRank={xpToNextRank}
        rank={profile.jediRank}
      />
      
      <S.Stats>
        <S.Stat>
          <S.StatValue>{profile.totalQueries}</S.StatValue>
          <S.StatLabel>Consultas</S.StatLabel>
        </S.Stat>
        <S.Stat>
          <S.StatValue>{profile.achievements.length}</S.StatValue>
          <S.StatLabel>Conquistas</S.StatLabel>
        </S.Stat>
      </S.Stats>
      
      {recentAchievements.length > 0 && (
        <S.RecentAchievements>
          <S.SectionTitle>🏆 Conquistas Recentes</S.SectionTitle>
          <AchievementGrid achievements={recentAchievements} />

  );
```
